cmake_minimum_required(VERSION 3.16)
project(PersonalFinanceTracker VERSION 1.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SQLite3 REQUIRED)
set(CORE_SOURCES
    src/core/Transaction.cpp src/core/Category.cpp
    src/services/FinanceManager.cpp src/services/CsvService.cpp src/services/ReportService.cpp
    src/storage/Database.cpp src/utils/Date.cpp src/utils/Format.cpp)
add_executable(personal_finance_tracker src/main.cpp ${CORE_SOURCES} src/ui/Console.cpp)
target_include_directories(personal_finance_tracker PRIVATE include)
target_link_libraries(personal_finance_tracker PRIVATE SQLite::SQLite3)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(personal_finance_tracker PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(BUILD_QT5_GUI "Build the Qt5 graphical application" ON)
if(BUILD_QT5_GUI)
  find_package(Qt5 COMPONENTS Widgets QUIET)
  if(Qt5Widgets_FOUND)
    set(CMAKE_AUTOMOC ON)
    set(CMAKE_AUTOUIC ON)
    set(CMAKE_AUTORCC ON)
    add_executable(personal_finance_tracker_qt src/qt/main_qt.cpp src/qt/MainWindow.cpp include/qt/MainWindow.hpp ${CORE_SOURCES})
    target_include_directories(personal_finance_tracker_qt PRIVATE include)
    target_link_libraries(personal_finance_tracker_qt PRIVATE Qt5::Widgets SQLite::SQLite3)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(personal_finance_tracker_qt PRIVATE -Wall -Wextra -Wpedantic)
    endif()
  else()
    message(WARNING "Qt5 Widgets not found; console target will still be built. Install qtbase5-dev to build the GUI.")
  endif()
endif()
enable_testing()
add_executable(finance_tests tests/test_finance.cpp ${CORE_SOURCES})
target_include_directories(finance_tests PRIVATE include)
target_link_libraries(finance_tests PRIVATE SQLite::SQLite3)
add_test(NAME finance_tests COMMAND finance_tests)
